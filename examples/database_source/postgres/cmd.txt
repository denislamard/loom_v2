
docker run -d \
  --name pg-data \
  -e POSTGRES_USER=postgres \
  -e POSTGRES_PASSWORD=postgres \
  -e POSTGRES_DB=data \
  -p 5433:5432 \
  -v pg-dev-data:/var/lib/postgresql \
  --restart unless-stopped \
  postgres:18
  
  

psql "postgresql://postgres:postgres@localhost:5433/data" -v ON_ERROR_STOP=1 -f schema.sql

psql "postgresql://postgres:postgres@localhost:5433/data" -v ON_ERROR_STOP=1 -f data.sql
